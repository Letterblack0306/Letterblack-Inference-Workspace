### 2. Create the Form Renderer

**`web/js/form-renderer.js`**

```javascript
// ============================================
// FORM RENDERER - Renders forms from registry
// ============================================

import { FORM_REGISTRY } from './form-registry.js';

class FormRenderer {
  constructor() {
    this.forms = FORM_REGISTRY;
    this.fieldTypes = {
      'text': this.renderText,
      'number': this.renderNumber,
      'port': this.renderPort,
      'url': this.renderUrl,
      'ip-address': this.renderIPAddress,
      'select': this.renderSelect,
      'checkbox': this.renderCheckbox,
      'checkbox-group': this.renderCheckboxGroup,
      'textarea': this.renderTextarea,
      'file-picker': this.renderFilePicker,
      'folder-picker': this.renderFolderPicker,
      'json-editor': this.renderJSONEditor,
      'text-list': this.renderTextList,
    };
  }

  render(formId, data = {}, options = {}) {
    const formDef = this.forms[formId];
    if (!formDef) {
      console.error(`Form not found: ${formId}`);
      return null;
    }

    const container = document.createElement('div');
    container.className = `form-renderer form-${formDef.layout || 'full'}`;

    // Render header
    container.innerHTML = this.renderHeader(formDef);

    // Render body
    const body = document.createElement('div');
    body.className = 'form-body';

    if (formDef.sections) {
      // Sectioned form (settings style)
      formDef.sections.forEach(section => {
        body.appendChild(this.renderSection(section, data));
      });
    } else {
      // Flat form
      body.appendChild(this.renderFields(formDef.fields, data));
    }

    container.appendChild(body);

    // Render footer
    if (!options.noFooter) {
      container.appendChild(this.renderFooter(formDef));
    }

    return container;
  }

  renderHeader(formDef) {
    return `
      <div class="form-header">
        ${formDef.icon ? `<svg class="form-icon"><use href="assets/icons.svg#${formDef.icon}"/></svg>` : ''}
        <h2>${formDef.title}</h2>
        <button class="form-close" type="button">✕</button>
      </div>
    `;
  }

  renderSection(section, data) {
    const container = document.createElement('div');
    container.className = 'form-section';
    container.innerHTML = `
      <div class="form-section-header">
        <h3>${section.label}</h3>
        <p>${section.description || ''}</p>
      </div>
      <div class="form-section-body">
        ${this.renderFields(section.fields, data)}
      </div>
    `;
    return container;
  }

  renderFields(fields, data) {
    const container = document.createElement('div');
    container.className = 'form-fields';

    fields.forEach(field => {
      const wrapper = this.renderField(field, data);
      if (wrapper) {
        if (field.span) {
          wrapper.style.gridColumn = `span ${field.span}`;
        }
        container.appendChild(wrapper);
      }
    });

    return container;
  }

  renderField(field, data) {
    const renderFn = this.fieldTypes[field.type];
    if (!renderFn) {
      console.warn(`Unknown field type: ${field.type}`);
      return null;
    }

    const wrapper = document.createElement('div');
    wrapper.className = `form-field field-${field.type}`;

    // Label
    if (field.type !== 'checkbox') {
      const label = document.createElement('label');
      label.textContent = field.label;
      if (field.required) {
        label.innerHTML += ' <span class="required">*</span>';
      }
      wrapper.appendChild(label);
    }

    // Input
    const input = renderFn(field, data);
    wrapper.appendChild(input);

    // Help text
    if (field.help) {
      const help = document.createElement('div');
      help.className = 'field-help';
      help.textContent = field.help;
      wrapper.appendChild(help);
    }

    // Validation message
    if (field.validation) {
      const msg = document.createElement('div');
      msg.className = 'field-validation';
      msg.hidden = true;
      wrapper.appendChild(msg);
    }

    return wrapper;
  }

  // ----- Field Type Renderers -----

  renderText(field, data) {
    const input = document.createElement('input');
    input.type = 'text';
    input.id = `field-${field.id}`;
    input.name = field.id;
    input.value = data[field.id] || field.default || '';
    input.placeholder = field.placeholder || '';
    input.readOnly = field.readonly || false;
    input.required = field.required || false;
    if (field.validation?.pattern) {
      input.pattern = field.validation.pattern;
    }
    return input;
  }

  renderNumber(field, data) {
    const input = document.createElement('input');
    input.type = 'number';
    input.id = `field-${field.id}`;
    input.name = field.id;
    input.value = data[field.id] || field.default || '';
    input.placeholder = field.placeholder || '';
    input.min = field.min;
    input.max = field.max;
    input.step = field.step || 1;
    input.required = field.required || false;
    return input;
  }

  renderPort(field, data) {
    const wrapper = document.createElement('div');
    wrapper.className = 'port-input';

    const input = document.createElement('input');
    input.type = 'number';
    input.id = `field-${field.id}`;
    input.name = field.id;
    input.value = data[field.id] || field.default || 8080;
    input.min = 1;
    input.max = 65535;
    input.required = true;
    wrapper.appendChild(input);

    const hint = document.createElement('span');
    hint.className = 'port-hint';
    hint.textContent = '1-65535';
    wrapper.appendChild(hint);

    return wrapper;
  }

  renderIPAddress(field, data) {
    const input = document.createElement('input');
    input.type = 'text';
    input.id = `field-${field.id}`;
    input.name = field.id;
    input.value = data[field.id] || '';
    input.placeholder = field.placeholder || '192.168.1.100';
    input.pattern = '^(\\d{1,3}\\.){3}\\d{1,3}$|^[a-zA-Z0-9.-]+$';
    input.required = field.required || false;
    input.setAttribute('data-validate', 'ip-address');
    return input;
  }

  renderUrl(field, data) {
    const input = document.createElement('input');
    input.type = 'url';
    input.id = `field-${field.id}`;
    input.name = field.id;
    input.value = data[field.id] || '';
    input.placeholder = field.placeholder || 'http://127.0.0.1:8088';
    input.required = field.required || false;
    return input;
  }

  renderSelect(field, data) {
    const select = document.createElement('select');
    select.id = `field-${field.id}`;
    select.name = field.id;
    select.required = field.required || false;

    field.options.forEach(option => {
      const opt = document.createElement('option');
      opt.value = option.value;
      opt.textContent = option.label;
      if ((data[field.id] || field.default) === option.value) {
        opt.selected = true;
      }
      select.appendChild(opt);
    });

    return select;
  }

  renderCheckbox(field, data) {
    const wrapper = document.createElement('label');
    wrapper.className = 'checkbox-wrapper';

    const input = document.createElement('input');
    input.type = 'checkbox';
    input.id = `field-${field.id}`;
    input.name = field.id;
    input.checked = data[field.id] !== undefined ? data[field.id] : (field.default || false);
    wrapper.appendChild(input);

    const span = document.createElement('span');
    span.textContent = field.label;
    wrapper.appendChild(span);

    return wrapper;
  }

  renderCheckboxGroup(field, data) {
    const container = document.createElement('div');
    container.className = 'checkbox-group';

    const options = field.options || data[`${field.id}_options`] || [];
    options.forEach(option => {
      const label = document.createElement('label');
      const input = document.createElement('input');
      input.type = 'checkbox';
      input.name = field.id;
      input.value = option.value || option;
      if (data[field.id]?.includes(input.value)) {
        input.checked = true;
      }
      label.appendChild(input);
      label.appendChild(document.createTextNode(option.label || option));
      container.appendChild(label);
    });

    return container;
  }

  renderTextarea(field, data) {
    const textarea = document.createElement('textarea');
    textarea.id = `field-${field.id}`;
    textarea.name = field.id;
    textarea.rows = field.rows || 4;
    textarea.placeholder = field.placeholder || '';
    textarea.value = data[field.id] || field.default || '';
    return textarea;
  }

  renderFilePicker(field, data) {
    const wrapper = document.createElement('div');
    wrapper.className = 'file-picker';

    const input = document.createElement('input');
    input.type = 'text';
    input.id = `field-${field.id}`;
    input.name = field.id;
    input.value = data[field.id] || '';
    input.placeholder = field.placeholder || 'Select a file...';
    wrapper.appendChild(input);

    const button = document.createElement('button');
    button.type = 'button';
    button.className = 'button secondary compact';
    button.textContent = 'Browse...';
    button.addEventListener('click', () => this.openFilePicker(field, input));
    wrapper.appendChild(button);

    return wrapper;
  }

  renderFolderPicker(field, data) {
    const wrapper = document.createElement('div');
    wrapper.className = 'file-picker';

    const input = document.createElement('input');
    input.type = 'text';
    input.id = `field-${field.id}`;
    input.name = field.id;
    input.value = data[field.id] || '';
    input.placeholder = field.placeholder || 'Select a folder...';
    wrapper.appendChild(input);

    const button = document.createElement('button');
    button.type = 'button';
    button.className = 'button secondary compact';
    button.textContent = 'Browse...';
    button.addEventListener('click', () => this.openFolderPicker(field, input));
    wrapper.appendChild(button);

    return wrapper;
  }

  renderJSONEditor(field, data) {
    const textarea = document.createElement('textarea');
    textarea.id = `field-${field.id}`;
    textarea.name = field.id;
    textarea.rows = field.rows || 8;
    textarea.spellcheck = false;
    textarea.className = 'json-editor';
    const value = data[field.id] || field.default || '{}';
    textarea.value = typeof value === 'object' ? JSON.stringify(value, null, 2) : value;
    return textarea;
  }

  renderTextList(field, data) {
    const textarea = document.createElement('textarea');
    textarea.id = `field-${field.id}`;
    textarea.name = field.id;
    textarea.rows = field.rows || 3;
    textarea.placeholder = field.placeholder || 'One item per line';
    const value = data[field.id] || field.default || [];
    textarea.value = Array.isArray(value) ? value.join('\n') : value;
    return textarea;
  }

  // ----- File/Folder Pickers -----

  openFilePicker(field, input) {
    // Use showOpenFilePicker if available (Chromium)
    if (window.showOpenFilePicker) {
      window.showOpenFilePicker({
        types: field.accept ? [{ accept: { 'application/*': field.accept.split(',') } }] : undefined,
      }).then(handles => {
        if (handles.length > 0) {
          input.value = handles[0].name;
          // Dispatch change event
          input.dispatchEvent(new Event('change'));
        }
      }).catch(() => {});
      return;
    }

    // Fallback: use hidden file input
    const fileInput = document.createElement('input');
    fileInput.type = 'file';
    fileInput.accept = field.accept || '';
    fileInput.onchange = (e) => {
      if (e.target.files.length > 0) {
        input.value = e.target.files[0].path || e.target.files[0].name;
        input.dispatchEvent(new Event('change'));
      }
      fileInput.remove();
    };
    fileInput.click();
  }

  openFolderPicker(field, input) {
    // Use showDirectoryPicker if available (Chromium)
    if (window.showDirectoryPicker) {
      window.showDirectoryPicker().then(handle => {
        input.value = handle.name;
        input.dispatchEvent(new Event('change'));
      }).catch(() => {});
      return;
    }

    // Fallback: use hidden file input with webkitdirectory
    const fileInput = document.createElement('input');
    fileInput.type = 'file';
    fileInput.webkitdirectory = true;
    fileInput.onchange = (e) => {
      if (e.target.files.length > 0) {
        const path = e.target.files[0].path || e.target.files[0].webkitRelativePath;
        input.value = path;
        input.dispatchEvent(new Event('change'));
      }
      fileInput.remove();
    };
    fileInput.click();
  }

  // ----- Validation -----

  validate(formId, data) {
    const formDef = this.forms[formId];
    if (!formDef) return { valid: false, errors: ['Form not found'] };

    const errors = [];
    const allFields = formDef.sections
      ? formDef.sections.flatMap(s => s.fields)
      : formDef.fields;

    allFields.forEach(field => {
      const value = data[field.id];
      const isCheckbox = field.type === 'checkbox';

      // Required check
      if (field.required && !isCheckbox) {
        if (!value || (typeof value === 'string' && !value.trim())) {
          errors.push({ field: field.id, message: `${field.label} is required` });
          return;
        }
      }

      // Pattern validation
      if (field.validation?.pattern && value) {
        const regex = new RegExp(field.validation.pattern);
        if (!regex.test(value)) {
          errors.push({
            field: field.id,
            message: field.validation.message || `${field.label} has invalid format`,
          });
        }
      }

      // Type-specific validation
      if (field.type === 'port') {
        const port = Number(value);
        if (port < 1 || port > 65535) {
          errors.push({ field: field.id, message: `${field.label} must be between 1-65535` });
        }
      }

      if (field.type === 'ip-address' && value) {
        const ipRegex = /^(\d{1,3}\.){3}\d{1,3}$/;
        const hostnameRegex = /^[a-zA-Z0-9.-]+$/;
        if (!ipRegex.test(value) && !hostnameRegex.test(value)) {
          errors.push({ field: field.id, message: `${field.label} must be a valid IP or hostname` });
        }
      }
    });

    return {
      valid: errors.length === 0,
      errors,
    };
  }

  // ----- Data Collection -----

  collect(formElement) {
    const form = formElement.querySelector('form') || formElement;
    const data = {};
    const inputs = form.querySelectorAll('[name]');

    inputs.forEach(input => {
      const name = input.name;
      if (input.type === 'checkbox') {
        if (input.closest('.checkbox-group')) {
          // Checkbox group - collect array
          if (!data[name]) data[name] = [];
          if (input.checked) data[name].push(input.value);
        } else {
          // Single checkbox - boolean
          data[name] = input.checked;
        }
      } else if (input.type === 'textarea' && input.classList.contains('json-editor')) {
        // JSON editor - parse JSON
        try {
          data[name] = JSON.parse(input.value);
        } catch {
          data[name] = input.value;
        }
      } else if (input.type === 'textarea' && input.classList.contains('text-list')) {
        // Text list - split by newlines
        data[name] = input.value.split('\n').filter(s => s.trim());
      } else {
        data[name] = input.value;
      }
    });

    return data;
  }

  // ----- Populate -----

  populate(formElement, data) {
    const form = formElement.querySelector('form') || formElement;
    const inputs = form.querySelectorAll('[name]');

    inputs.forEach(input => {
      const name = input.name;
      if (data[name] !== undefined) {
        if (input.type === 'checkbox') {
          if (input.closest('.checkbox-group')) {
            input.checked = data[name]?.includes(input.value);
          } else {
            input.checked = Boolean(data[name]);
          }
        } else if (input.type === 'textarea' && input.classList.contains('json-editor')) {
          input.value = typeof data[name] === 'object'
            ? JSON.stringify(data[name], null, 2)
            : data[name];
        } else if (input.type === 'textarea' && input.classList.contains('text-list')) {
          input.value = Array.isArray(data[name]) ? data[name].join('\n') : data[name];
        } else {
          input.value = data[name];
        }
      }
    });
  }
}

export const formRenderer = new FormRenderer();
```
